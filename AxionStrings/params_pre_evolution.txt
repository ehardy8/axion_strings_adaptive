# AxionStrings parameter file -- pre-evolution stage of the requested run:
# N=512 (box size), N1=1 (HL=1 at final time), N2=1 (m_r*Delta=1 at final
# time), fat string (c0=1), starting the main run at log(m_r/H_i)=2,
# relaxing a Fourier-mode IC (modes up to m_r/2, mean-square amplitude v)
# down to xi=0.7 before handoff.
#
# Workflow: run this file first (produces a checkpoint under ./checkpoints/
# once xi reaches the target), then point params_main.txt's amr.restart at
# that checkpoint and run that second.

#################################################
# Filesystem parameters

amr.verbose = 1
# >=0 is required for pre-evolution (AxionStringsParams::check_params
# enforces this): the whole point is to hand off via a checkpoint. A large
# value effectively means "only the guaranteed final checkpoint, no
# periodic mid-run ones" -- Main_AxionStrings.cpp always writes a final
# checkpoint once the xi-monitoring loop stops, regardless of this value,
# as long as it is >= 0.
amr.check_int = 1000000
amr.plot_int  = -1

#################################################
# Grid parameters -- amr.n_cell/geometry.prob_extent are NOT set here:
# box planning below derives them from axion_strings.N/N1/N2 (printed at
# startup for you to sanity-check). Expect geometry.prob_extent = 512 512
# 512 (comoving) and amr.n_cell = 512 512 512.

amr.max_level = 0 # single-level throughout milestone 1

# TODO: tune to your actual node/rank allocation. 512 = 8 * 64, so this
# allows domain decomposition into up to 8^3 = 512 boxes.
amr.max_grid_size   = 64
amr.blocking_factor = 32
amr.n_error_buf     = 1

#################################################
# Boundary condition parameters

geometry.is_periodic = 1 1 1

#################################################
# Evolution parameters

evolution.dt_multiplier = 0.2
# Pre-evolution stops via the xi-monitoring loop, not a fixed stop_time --
# but GRTeclyn's own BaseParameterChecker silently defaults
# evolution.stop_time to 1.0 if left unset (independent of
# AxionStringsParams::apply_box_plan, which deliberately does not set it
# for pre-evolution -- found this the hard way, by testing: without this
# line the run above silently stopped at t=1.0, not at the xi target).
# -1 means unlimited, matching max_steps' own convention below.
evolution.stop_time = -1
# max_steps is the real safety cap for this run, in case xi never reaches
# the target; generous on purpose.
evolution.max_steps = 200000

#################################################
# AxionStrings parameters (conventions.md sec.2, 4, 5, 7)

axion_strings.mode = pre_evolution

# a_inv = 2 <-> radiation domination (b_inv = 1)
axion_strings.a_inv = 2.0

# c0 = 1 <-> fat string ("for the fat string system")
axion_strings.c0 = 1.0

# log(m_r/H) at the *main run's* start -- this run's own clock is separate
# (tau_pre, PreEvolutionBackground.hpp), but tau_i derived from this is
# still used here: it's the time at which the xi_target below is measured
# (conventions.md sec.7 -- xi is targeted at the main run's start, not at
# this run's own current time), and it feeds box planning below (must
# match params_main.txt exactly).
axion_strings.log_mr_over_h_i = 2.0

# Box planning (conventions.md sec.5): N = box size, N1 = H*L at the final
# time ("up to HL=1"), N2 = 1/(Delta*m_r) at the final time ("m_r Delta=1").
# Must match params_main.txt exactly -- both runs need the same comoving
# box (geometry.prob_extent), derived from the same N/N1/N2/a_inv/c0.
# With these numbers: tau_f = 512, L_tilde (comoving) = 512, dx (comoving)
# = 1, tau_i = e^2 = 7.389, so this run's own xi target is measured as if
# at tau = 7.389 (verify these printed numbers match at startup).
axion_strings.N  = 512
axion_strings.N1 = 1.0
axion_strings.N2 = 1.0

# Target (constant) m_r/H to relax the network at during pre-evolution
# (PreEvolutionBackground.hpp) -- NOT specified in the request, so this is
# a placeholder: e^(log_mr_over_h_i) = m_r/H_i for the main run, on the
# reasoning that pre-evolution's own resolution regime should be
# comparable to what the main run starts at. Please sanity-check/override.
axion_strings.pre_evolution.gamma = 7.389

# Stop once the measured xi (at tau_i, per the log_mr_over_h_i note above)
# first drops to this value.
axion_strings.pre_evolution.xi_target = 0.7

# Fourier-mode IC (conventions.md sec.7): modes populated for |k| <= this
# times m_r ("modes up to m_r/2").
axion_strings.pre_evolution.k_max_over_mr = 0.5

# NOTE on this value -- please confirm: FourierIC.hpp targets this as the
# mean square of EACH of psi1, psi2 independently (see FourierIC.hpp's
# `generate`, called once per component with the same target). So
# <|psi|^2> = <psi1^2> + <psi2^2> = 2 * mean_square_variance. For "a mean
# square field amplitude of v" (v=1 in code units, i.e. <|psi|^2> = v^2 =
# 1), that means mean_square_variance = 0.5, NOT 1.0 -- set here
# accordingly. If you actually meant <psi1^2> = <psi2^2> = v^2 = 1 (a
# different convention), use 1.0 instead.
axion_strings.pre_evolution.mean_square_variance = 0.5

# RNG seed (CLAUDE.md constraint 7: fixed for reproducibility). Change this
# per ensemble realisation.
axion_strings.pre_evolution.seed = 12345
