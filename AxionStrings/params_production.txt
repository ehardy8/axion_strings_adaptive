# AxionStrings parameters (conventions.md sec.2, 4, 5, 7)

# a_inv = 2 <-> radiation domination (b_inv = 1)
axion_strings.a_inv = 2.0

# c0 = 1 <-> fat string ("for the fat string system")
axion_strings.c0 = 1.0

# log(m_r/H) at the *main evolution's* start -- tau_i is derived from this.
# This is also when the xi_target below is measured (conventions.md sec.7:
# xi is targeted at the main run's start, not at the relaxation phase's own
# current time).
axion_strings.log_mr_over_h_i = 2.0

# Box planning (conventions.md sec.5): N = box size, N1 = H*L at the final
# time ("up to HL=1"), N2 = 1/(Delta*m_r) at the final time ("m_r Delta=1").
# With these numbers: tau_f = 512, L_tilde (comoving) = 512, dx (comoving)
# = 1, tau_i = e^2 = 7.389 (verify these printed numbers match at startup).
axion_strings.N  = 512
axion_strings.N1 = 1.0
axion_strings.N2 = 1.0

# Relax a Fourier-mode IC down to xi_target before continuing to the main
# evolution, all in this one run (AxionStringsLevel::Phase).
axion_strings.ic_mode = fourier_relaxed

# Target (constant) m_r/H to relax the network at during the relaxation
# phase (PreEvolutionBackground.hpp) -- NOT specified in the request, so
# this is a placeholder: e^(log_mr_over_h_i) = m_r/H_i for the main run, on
# the reasoning that the relaxation phase's own resolution regime should be
# comparable to what the main evolution starts at. Please sanity-check/
# override.
axion_strings.pre_evolution.gamma = 7.389

# Stop relaxing once the measured xi (at tau_i, per the log_mr_over_h_i
# note above) first drops to this value.
axion_strings.pre_evolution.xi_target = 0.7

# Fourier-mode IC (conventions.md sec.7): modes populated for |k| <= this
# times m_r ("modes up to m_r/2").
axion_strings.pre_evolution.k_max_over_mr = 0.5

# NOTE on this value -- please confirm: FourierIC.hpp targets this as the
# mean square of EACH of psi1, psi2 independently (see FourierIC.hpp's
# `generate`, called once per component with the same target). So
# <|psi|^2> = <psi1^2> + <psi2^2> = 2 * mean_square_variance. For "a mean
# square field amplitude of v" (v=1 in code units, i.e. <|psi|^2> = v^2 =
# 1), that means mean_square_variance = 0.5, NOT 1.0 -- set here
# accordingly. If you actually meant <psi1^2> = <psi2^2> = v^2 = 1 (a
# different convention), use 1.0 instead.
axion_strings.pre_evolution.mean_square_variance = 0.5

# RNG seed (CLAUDE.md constraint 7: fixed for reproducibility). Change this
# per ensemble realisation.
axion_strings.pre_evolution.seed = 12345
